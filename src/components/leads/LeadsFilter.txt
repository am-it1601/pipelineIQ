import {
    Field,
    FieldLabel
} from "@/components/ui/field";
import {
    InputGroup,
    InputGroupAddon,
    InputGroupInput
} from "@/components/ui/input-group";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { BID_TYPES, ENGAGEMENT_TYPES, LEAD_SOURCES, STATUSES } from "@/lib/constants";
import type { BDMember, UpworkProfile } from "@/lib/types";
import { SearchIcon, Star } from "lucide-react";
import { Button } from "../ui/button";

const LeadsFilter = (
    {
        setPage, isAdmin, profiles, members,
        search, setSearch,
        filterStatus, setFilterStatus,
        filterSource, setFilterSource,
        filterMember, setFilterMember,
        filterEngagement, setFilterEngagement,
        filterBidType, setFilterBidType,
        filterProfile, setFilterProfile,
        filterHot, setFilterHot,
    }: {
        setPage: (page: number) => void;
        isAdmin: boolean;
        profiles: UpworkProfile[];
        members: BDMember[];
        search: string;
        setSearch: (val: string) => void;
        filterStatus: string;
        setFilterStatus: (val: string) => void;
        filterSource: string;
        setFilterSource: (val: string) => void;
        filterMember: string;
        setFilterMember: (val: string) => void;
        filterEngagement: string;
        setFilterEngagement: (val: string) => void;
        filterBidType: string;
        setFilterBidType: (val: string) => void;
        filterProfile: string;
        setFilterProfile: (val: string) => void;
        filterHot: boolean;
        setFilterHot: (val: boolean) => void;
    }
) => {
    const resetPage = () => setPage(1);
    const activeFilterCount = [filterStatus, filterSource, filterMember, filterEngagement, filterBidType, filterProfile, filterHot ? 'hot' : ''].filter(Boolean).length;

    return (

        <div className="grid grid-cols-4 grid-auto-flow auto-rows-auto gap-4">
            <Field>
                <FieldLabel htmlFor="input-group-url">Search</FieldLabel>
                <InputGroup>
                    <InputGroupInput id="input-group-url"
                        onChange={(e) => { setSearch(e.target.value); resetPage(); }} />
                    <InputGroupAddon align="inline-start">
                        <SearchIcon />
                    </InputGroupAddon>
                </InputGroup>
            </Field>
            <Field>
                <FieldLabel htmlFor='lead_status'>Status</FieldLabel>
                <Select value={filterStatus || 'all'} onValueChange={(v) => { setFilterStatus(v === 'all' || !v ? '' : v); resetPage(); }}>
                    <SelectTrigger id='lead_status'><SelectValue placeholder="All Statuses" /></SelectTrigger>
                    <SelectContent><SelectItem value="all">All Statuses</SelectItem>{STATUSES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
                </Select>
            </Field>
            <Field>
                <FieldLabel htmlFor='lead_source'>Source</FieldLabel>
                <Select value={filterSource || 'all'} onValueChange={(v) => { setFilterSource(v === 'all' || !v ? '' : v); resetPage(); }}>
                    <SelectTrigger id='lead_source'><SelectValue placeholder="All Sources" /></SelectTrigger>
                    <SelectContent><SelectItem value="all">All Sources</SelectItem>
                        {LEAD_SOURCES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
                </Select>
            </Field>
            <Field>
                <FieldLabel htmlFor='lead_eng_type'>Engagement Type</FieldLabel>
                <Select value={filterEngagement || 'all'} onValueChange={(v) => { setFilterEngagement(v === 'all' || !v ? '' : v); resetPage(); }}>
                    <SelectTrigger id='lead_eng_type'><SelectValue placeholder="All Engagement Types" /></SelectTrigger>
                    <SelectContent><SelectItem value="all">All Engagement Types</SelectItem>
                        {ENGAGEMENT_TYPES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
                </Select>
            </Field>
            <Field>
                <FieldLabel htmlFor='lead_bid_type'>Bid Type</FieldLabel>
                <Select value={filterBidType || 'all'} onValueChange={(v) => { setFilterBidType(v === 'all' || !v ? '' : v); resetPage(); }}>
                    <SelectTrigger id='lead_bid_type'><SelectValue placeholder="All Bid Types" /></SelectTrigger>
                    <SelectContent><SelectItem value="all">All Bid Types</SelectItem>
                        {BID_TYPES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
                </Select>
            </Field>
            <Field>
                <FieldLabel htmlFor='lead_profiles'>Profile</FieldLabel>
                <Select value={filterProfile || 'all'} onValueChange={(v) => { setFilterProfile(v === 'all' || !v ? '' : v); resetPage(); }}>
                    <SelectTrigger id='lead_profiles'><SelectValue placeholder="All Profiles" /></SelectTrigger>
                    <SelectContent><SelectItem value="all">All Profiles</SelectItem>{profiles.map(p => <SelectItem key={p.id} value={p.id}>{p.profile_name}</SelectItem>)}</SelectContent>                        </Select>
            </Field>
            {
                isAdmin && (
                    <Field>
                        <FieldLabel htmlFor='lead_assignee'>Assignee</FieldLabel>
                        <Select value={filterMember || 'all'} onValueChange={(v) => { setFilterMember(v === 'all' || !v ? '' : v); resetPage(); }}>
                            <SelectTrigger id='lead_assignee'><SelectValue placeholder="All Members" /></SelectTrigger>
                            <SelectContent><SelectItem value="all">All Members</SelectItem>{members.map(m => <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>)}</SelectContent>
                        </Select>
                    </Field>
                )
            }
            <Field className='w-24'>
                <FieldLabel htmlFor='isStarred'>Starred</FieldLabel>
                <Button id='isStarred' variant={filterHot ? "default" : "outline"} size="sm" className={`h-9 text-xs ${filterHot ? 'bg-amber-500 hover:bg-amber-600' : ''}`} onClick={() => { setFilterHot(!filterHot); resetPage(); }}>
                    <Star className={`w-3.5 h-3.5 mr-1 ${filterHot ? 'fill-current' : ''}`} />
                    Starred
                </Button>
            </Field>
        </div>

    )
}

export default LeadsFilter
