'use client';

import { Field, FieldLabel } from "@/components/ui/field";
import { InputGroup, InputGroupAddon, InputGroupInput } from "@/components/ui/input-group";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { BID_TYPES, ENGAGEMENT_TYPES, LEAD_SOURCES, STATUSES } from "@/lib/constants";
import type { BDMember, UpworkProfile } from "@/lib/types";
import { SearchIcon, Star } from "lucide-react";
import { Button } from "../ui/button";

interface LeadsFilterClientProps {
  members: BDMember[];
  profiles: UpworkProfile[];
  isAdmin: boolean;
  onFilterChange: (filters: Record<string, string | number | boolean>) => void;
  currentFilters: Record<string, string | number | boolean>;
}

// Helper function to ensure value is string
const ensureString = (value: any): string => {
  if (typeof value === 'string') return value;
  if (typeof value === 'number' || typeof value === 'boolean') return String(value);
  return '';
};

export default function LeadsFilterClient({
  members,
  profiles,
  isAdmin,
  onFilterChange,
  currentFilters,
}: LeadsFilterClientProps) {
  const search: string = ensureString(currentFilters.search || '');
  const filterStatus: string = ensureString(currentFilters.status || 'all');
  const filterSource: string = ensureString(currentFilters.source || 'all');
  const filterMember: string = ensureString(currentFilters.member || 'all');
  const filterEngagement: string = ensureString(currentFilters.engagement || 'all');
  const filterBidType: string = ensureString(currentFilters.bidType || 'all');
  const filterProfile: string = ensureString(currentFilters.profile || 'all');
  const filterHot: boolean = Boolean(currentFilters.hot);

  return (
    <div className="grid grid-cols-4 grid-auto-flow auto-rows-auto gap-4 py-4">
      <Field>
        <FieldLabel htmlFor="search-input">Search</FieldLabel>
        <InputGroup>
          <InputGroupInput
            id="search-input"
            value={search}
            onChange={(e) => { onFilterChange({ search: e.target.value, page: 1 }); }}
            placeholder="Project name..."
          />
          <InputGroupAddon align="inline-start">
            <SearchIcon />
          </InputGroupAddon>
        </InputGroup>
      </Field>
      
      <Field>
        <FieldLabel htmlFor='lead_status'>Status</FieldLabel>
        {/* @ts-ignore - filterStatus is always string, Select type inference issue */}
        <Select value={filterStatus} onValueChange={(v) => onFilterChange({ status: v === 'all' ? '' : v, page: 1 })}>
          <SelectTrigger id='lead_status'><SelectValue placeholder="All Statuses" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">All Statuses</SelectItem>
            {STATUSES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}
          </SelectContent>
        </Select>
      </Field>
      
      <Field>
        <FieldLabel htmlFor='lead_source'>Source</FieldLabel>
        {/* @ts-ignore - filterSource is always string, Select type inference issue */}
        <Select value={filterSource} onValueChange={(v) => onFilterChange({ source: v === 'all' ? '' : v, page: 1 })}>

          <SelectTrigger id='lead_source'><SelectValue placeholder="All Sources" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">All Sources</SelectItem>
            {LEAD_SOURCES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}
          </SelectContent>
        </Select>
      </Field>
      
      <Field>
        <FieldLabel htmlFor='lead_eng_type'>Engagement Type</FieldLabel>
        {/* @ts-ignore - filterEngagement is always string, Select type inference issue */}
        <Select value={filterEngagement} onValueChange={(v) => onFilterChange({ engagement: v === 'all' ? '' : v, page: 1 })}>

          <SelectTrigger id='lead_eng_type'><SelectValue placeholder="All Types" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">All Types</SelectItem>
            {ENGAGEMENT_TYPES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}
          </SelectContent>
        </Select>
      </Field>
      
      <Field>
        <FieldLabel htmlFor='lead_bid_type'>Bid Type</FieldLabel>
        {/* @ts-ignore - filterBidType is always string, Select type inference issue */}
        <Select value={filterBidType} onValueChange={(v) => onFilterChange({ bidType: v === 'all' ? '' : v, page: 1 })}>

          <SelectTrigger id='lead_bid_type'><SelectValue placeholder="All Types" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">All Types</SelectItem>
            {BID_TYPES.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}
          </SelectContent>
        </Select>
      </Field>
      
      <Field>
        <FieldLabel htmlFor='lead_profiles'>Profile</FieldLabel>
        {/* @ts-ignore - filterProfile is always string, Select type inference issue */}
        <Select value={filterProfile} onValueChange={(v) => onFilterChange({ profile: v === 'all' ? '' : v, page: 1 })}>
          <SelectTrigger id='lead_profiles'><SelectValue placeholder="All Profiles" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">All Profiles</SelectItem>
            {profiles.map(p => <SelectItem key={p.id} value={p.id}>{p.profile_name}</SelectItem>)}
          </SelectContent>
        </Select>
      </Field>
      
      {isAdmin && (
        <Field>
          <FieldLabel htmlFor='lead_assignee'>Assignee</FieldLabel>
          {/* @ts-ignore - filterMember is always string, Select type inference issue */}
          <Select value={filterMember} onValueChange={(v) => onFilterChange({ member: v === 'all' ? '' : v, page: 1 })}>

            <SelectTrigger id='lead_assignee'><SelectValue placeholder="All Members" /></SelectTrigger>
            <SelectContent>
              <SelectItem value="all">All Members</SelectItem>
              {members.map(m => <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>)}
            </SelectContent>
          </Select>
        </Field>
      )}
      
      <Field className='w-24'>
        <FieldLabel htmlFor='isHot'>Starred</FieldLabel>
        <Button
          id='isHot'
          variant={filterHot ? "default" : "outline"}
          size="sm"
          className={`h-9 text-xs ${filterHot ? 'bg-amber-500 hover:bg-amber-600' : ''}`}
          onClick={() => onFilterChange({ hot: !filterHot, page: 1 })}
        >
          <Star className={`w-3.5 h-3.5 mr-1 ${filterHot ? 'fill-current' : ''}`} />
          Starred
        </Button>
      </Field>
    </div>
  );
}
