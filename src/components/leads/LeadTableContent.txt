'use client';

import { useSearchParams, useRouter } from 'next/navigation';
import { useEffect, useState } from 'react';
import { useAuth } from '@/lib/hooks/useAuth';
import type { BDMember, LeadLogEntry, UpworkProfile } from '@/lib/types';
import LeadLogTableClient from './LeadLogTableClient';

interface LeadTableContentProps {
  initialLeads: LeadLogEntry[];
  pagination: {
    total: number;
    page: number;
    pageSize: number;
    totalPages: number;
  };
  members: BDMember[];
  profiles: UpworkProfile[];
}

export default function LeadTableContent({
  initialLeads,
  pagination: initialPagination,
  members,
  profiles,
}: LeadTableContentProps) {
  const router = useRouter();
  const searchParams = useSearchParams();
  const user = useAuth();
  
  const [leads, setLeads] = useState<LeadLogEntry[]>(initialLeads);
  const [pagination, setPagination] = useState(initialPagination);
  const [loading, setLoading] = useState(false);

  if (!user) return null;

  // Extract filter values from URL
  const currentFilters = {
    search: searchParams.get('search') || '',
    status: searchParams.get('status') || '',
    source: searchParams.get('source') || '',
    member: searchParams.get('member') || '',
    engagement: searchParams.get('engagement') || '',
    bidType: searchParams.get('bidType') || '',
    profile: searchParams.get('profile') || '',
    hot: searchParams.get('hot') === 'true',
    sortBy: searchParams.get('sortBy') || 'date',
    sortDir: (searchParams.get('sortDir') || 'desc') as 'asc' | 'desc',
    page: parseInt(searchParams.get('page') || '1', 10),
    pageSize: parseInt(searchParams.get('pageSize') || '20', 10),
  };

  // Refetch leads when URL search params change
  useEffect(() => {
    const fetchLeads = async () => {
      setLoading(true);
      try {
        const queryParams = new URLSearchParams();
        
        // Build query string from current filters
        if (currentFilters.search) queryParams.set('search', currentFilters.search);
        if (currentFilters.status) queryParams.set('status', currentFilters.status);
        if (currentFilters.source) queryParams.set('source', currentFilters.source);
        if (currentFilters.member) queryParams.set('member', currentFilters.member);
        if (currentFilters.engagement) queryParams.set('engagement', currentFilters.engagement);
        if (currentFilters.bidType) queryParams.set('bidType', currentFilters.bidType);
        if (currentFilters.profile) queryParams.set('profile', currentFilters.profile);
        if (currentFilters.hot) queryParams.set('hot', 'true');
        
        queryParams.set('sortBy', currentFilters.sortBy);
        queryParams.set('sortDir', currentFilters.sortDir);
        queryParams.set('page', String(currentFilters.page));
        queryParams.set('pageSize', String(currentFilters.pageSize));

        const response = await fetch(`/api/leads?${queryParams.toString()}`);
        if (!response.ok) throw new Error('Failed to fetch leads');
        
        const result = await response.json();
        setLeads(result.data || []);
        setPagination(result.pagination || initialPagination);
      } catch (error) {
        console.error('Error fetching leads:', error);
        // Keep existing data on error
      } finally {
        setLoading(false);
      }
    };

    fetchLeads();
  }, [searchParams]);

  // Handler to update filters and push to URL
  const handleFilterChange = (newFilters: Record<string, string | number | boolean>) => {
    const params = new URLSearchParams(searchParams);

    Object.entries(newFilters).forEach(([key, value]) => {
      if (value === '' || value === 0 || value === false) {
        params.delete(key);
      } else {
        params.set(key, String(value));
      }
    });

    // Always reset to page 1 when filters change (unless pagination-specific)
    if (Object.keys(newFilters).some(k => !['page', 'pageSize', 'sortBy', 'sortDir'].includes(k))) {
      params.set('page', '1');
    }

    router.push(`?${params.toString()}`, { scroll: false });
  };

  return (
    <LeadLogTableClient
      leads={leads}
      members={members}
      profiles={profiles}
      user={user}
      pagination={pagination}
      onFilterChange={handleFilterChange}
      currentFilters={currentFilters}
    />
  );
}
